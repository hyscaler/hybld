import { Card, CardGrid, LinkCard } from "@astrojs/starlight/components";

<CardGrid stagger>
  <Card title="Incremental by design" icon="rocket">
    A SQLite [manifest](/guides/concepts/) remembers every built page. New article? One build. Edited article? One build plus its [listing pages](/guides/concepts/#dependents). Deleted? Removed, with redirects for renames and the sitemap kept current.
  </Card>
  <Card title="Your framework, your renderer" icon="puzzle">
    [React + Vite](/frameworks/react-vite/) renders through hybld's own engine with millisecond warm rebuilds. [Next.js](/frameworks/nextjs/) and [Astro](/frameworks/astro/) keep their own builds; hybld scopes them to just the changed pages and merges the output.
  </Card>
  <Card title="Any CMS" icon="setting">
    Presets for [WordPress, Strapi, Contentful, and Sanity](/reference/adapters/), generic REST and GraphQL helpers for everything else, all behind one rate-limited, retrying, memoizing HTTP layer that never hammers your CMS.
  </Card>
  <Card title="Operations included" icon="approve-check">
    A [webhook server](/operate/deployment/) with ISR-style regeneration, precise [deploy diffs](/reference/cli/#hybld-sync-targetdir---delete---redirects-netlifynginx), chunked CI full builds, sharded sitemaps, RSS, and asset garbage collection.
  </Card>
</CardGrid>
